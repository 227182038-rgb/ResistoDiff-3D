  1130→// --- MCTS Tree helpers ---
  1131→function countNodes(node: MCTSNode): number {
  1132→  return 1 + node.children.reduce((sum, c) => sum + countNodes(c), 0);
  1133→}
  1134→
  1135→function countResistanceNodes(node: MCTSNode): number {
  1136→  return (
  1137→    (node.isResistance ? 1 : 0) +
  1138→    node.children.reduce((sum, c) => sum + countResistanceNodes(c), 0)
  1139→  );
  1140→}
  1141→
  1142→function maxDepth(node: MCTSNode): number {
  1143→  if (node.children.length === 0) return node.depth;
  1144→  return Math.max(...node.children.map(maxDepth));
  1145→}
  1146→
  1147→function bestFitness(node: MCTSNode): number {
  1148→  const childBest = node.children.length > 0
  1149→    ? Math.max(...node.children.map(bestFitness))
  1150→    : -Infinity;
  1151→  return Math.max(node.fitnessScore, childBest);
  1152→}

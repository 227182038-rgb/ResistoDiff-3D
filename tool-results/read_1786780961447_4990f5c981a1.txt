    88→
    89→class _TaskStore:
    90→    """In-memory store for background task status and results."""
    91→
    92→    def __init__(self) -> None:
    93→        self._tasks: Dict[str, Dict[str, Any]] = {}
    94→
    95→    def create(self, name: str) -> str:
    96→        task_id = uuid.uuid4().hex[:12]
    97→        self._tasks[task_id] = {
    98→            "task_id": task_id,
    99→            "name": name,
   100→            "status": "pending",
   101→            "created_at": time.time(),
   102→            "result": None,
   103→            "error": None,
   104→        }
   105→        return task_id
   106→
   107→    def update(self, task_id: str, status: str, result: Any = None, error: Any = None) -> None:
   108→        task = self._tasks.get(task_id)
   109→        if task is None:
   110→            return
   111→        task["status"] = status
   112→        task["completed_at"] = time.time()
   113→        if result is not None:
   114→            task["result"] = result
   115→        if error is not None:
   116→            task["error"] = error
   117→
   118→    def get(self, task_id: str) -> Optional[Dict[str, Any]]:
   119→        return self._tasks.get(task_id)
   120→

   390→def _resolve_pdb_path(target_id: str, pdb_id: Optional[str]) -> Path:
   391→    """Resolve the PDB file path for a target, downloading if needed.
   392→
   393→    Raises HTTPException if no PDB is available.
   394→    """
   395→    dm = get_dataset_manager()
   396→    target = dm.get_target(target_id)
   397→    if target is None:
   398→        raise HTTPException(status_code=404, detail=f"Target '{target_id}' not found.")
   399→
   400→    chosen_pdb = (pdb_id or target.pdb_ids[0]).upper()
   401→    path = dm.get_pdb_path(target_id, chosen_pdb)
   402→
   403→    if path is None:
   404→        # Try fetching from RCSB
   405→        path = dm.fetch_pdb_from_rcsb(chosen_pdb)
   406→
   407→    if path is None or not path.exists():
   408→        raise HTTPException(
   409→            status_code=422,
   410→            detail=f"No PDB file available for target '{target_id}' / PDB '{chosen_pdb}'.",
   411→        )
   412→    return path

   505→def _run_sefe_mutate(
   506→    task_id: str,
   507→    target_id: str,
   508→    mutations: List[Dict[str, Any]],
   509→    pdb_id: Optional[str],
   510→    scorer: MutationScorer,
   511→) -> None:
   512→    """Run SEFE mutation analysis in a background thread."""
   513→    try:
   514→        task_store.update(task_id, status="running")
   515→
   516→        pdb_path = _resolve_pdb_path(target_id, pdb_id)
   517→        graph_builder = ProteinGraphBuilder()
   518→        h, x, edge_index, res_info = graph_builder.parse_pdb_to_graph(pdb_path)
   519→
   520→        dm = get_dataset_manager()
   521→        target = dm.get_target(target_id)
   522→        if target is None:
   523→            raise ValueError(f"Target '{target_id}' not found.")
   524→
   525→        # Resolve active site residues for this target
   526→        active_site_resseqs: List[int] = []
   527→        for ann in target.pocket_annotations:
   528→            for res_str in ann.residues:
   529→                # Parse e.g. "SER70" → 70
   530→                try:
   531→                    active_site_resseqs.append(int(res_str[3:]))
   532→                except (ValueError, IndexError):
   533→                    pass
   534→        active_site_resseqs = list(set(active_site_resseqs))
   535→
   536→        # Map resseqs to graph indices
   537→        active_site_indices = [
   538→            i for i, info in enumerate(res_info) if info["resseq"] in active_site_resseqs
   539→        ]
   540→        active_mask = torch.zeros(h.size(0), dtype=torch.bool)
   541→        if active_site_indices:
   542→            for idx in active_site_indices:
   543→                if 0 <= idx < h.size(0):
   544→                    active_mask[idx] = True
   545→
   546→        # Catalytic residues (first 3 from first pocket annotation)
   547→        catalytic_indices: List[int] = []
   548→        for ann in target.pocket_annotations:
   549→            if ann.name in ("active_site", "zinc_binding_site_1"):
   550→                for res_str in ann.residues[:3]:
   551→                    try:
   552→                        resseq = int(res_str[3:])
   553→                        for i, info in enumerate(res_info):
   554→                            if info["resseq"] == resseq:
   555→                                catalytic_indices.append(i)
   556→                                break
   557→                    except (ValueError, IndexError):
   558→                        pass
   559→                if catalytic_indices:
   560→                    break
   561→
   562→        results: List[Dict[str, Any]] = []
   563→        for mut in mutations:
   564→            pos = mut["position"]
   565→            new_aa = mut["mutant"]
   566→
   567→            # Find graph index for this position
   568→            graph_idx = None
   569→            for i, info in enumerate(res_info):
   570→                if info["resseq"] == pos:
   571→                    graph_idx = i
   572→                    break
   573→
   574→            if graph_idx is None:
   575→                results.append({
   576→                    "mutation": mut,
   577→                    "error": f"Residue position {pos} not found in PDB structure.",
   578→                })
   579→                continue
   580→
   581→            # Generate mutant coordinates
   582→            mut_coords = generate_mutant_coords(x, graph_idx, new_aa)
   583→
   584→            # Compute scores
   585→            ddg_fold = scorer.compute_ddg_fold(x, mut_coords, edge_index, h)
   586→            ddg_bind = scorer.compute_ddg_binding(x, mut_coords, active_mask)
   587→            ddg_cat = scorer.compute_catalysis_effect(mut_coords, catalytic_indices)
   588→            fitness = scorer.compute_fitness_score(ddg_fold, ddg_cat, ddg_bind)
   589→            rmsd = 0.0
   590→            if graph_idx < x.size(0):
   591→                rmsd = float(torch.norm(x[graph_idx] - mut_coords[graph_idx]).item())
   592→
   593→            results.append({
   594→                "mutation": mut,
   595→                "ddg_fold": round(ddg_fold, 4),
   596→                "ddg_binding": round(ddg_bind, 4),
   597→                "catalysis_effect": round(ddg_cat, 4),
   598→                "fitness_score": round(fitness, 4),
   599→                "rmsd_wt_vs_mut": round(rmsd, 4),
   600→            })
   601→
   602→        task_store.update(task_id, status="completed", result=results)
   603→    except Exception as exc:
   604→        logger.exception("SEFE mutate task %s failed", task_id)
   605→        task_store.update(task_id, status="failed", error=str(exc))

   608→def _run_pocket_detection(
   609→    task_id: str,
   610→    target_id: str,
   611→    pdb_id: Optional[str],
   612→    detector: PocketDetector,
   613→) -> None:
   614→    """Run pocket detection in a background thread."""
   615→    try:
   616→        task_store.update(task_id, status="running")
   617→
   618→        pdb_path = _resolve_pdb_path(target_id, pdb_id)
   619→
   620→        # Parse PDB to get CA positions, radii, and residue IDs
   621→        from Bio.PDB import PDBParser
   622→
   623→        parser = PDBParser(QUIET=True)
   624→        structure = parser.get_structure("prot", str(pdb_path))
   625→        model = structure[0]
   626→
   627→        positions: List[List[float]] = []
   628→        radii: List[float] = []
   629→        residue_ids: List[str] = []
   630→        vdw_map = get_vdw_radii()
   631→
   632→        for chain in model:
   633→            for res in chain:
   634→                if res.id[0] != " ":
   635→                    continue
   636→                ca = res.get("CA")
   637→                if ca is None:
   638→                    continue
   639→                positions.append(ca.get_coord().tolist())
   640→                rname = res.get_resname()
   641→                radii.append(vdw_map.get(rname, 3.5))
   642→                residue_ids.append(f"{chain.id}:{res.id[1]}:{rname}")
   643→
   644→        if not positions:
   645→            raise ValueError("No CA atoms found in PDB file.")
   646→
   647→        pockets = detector.detect_pockets(
   648→            np.array(positions, dtype=np.float64),
   649→            np.array(radii, dtype=np.float64),
   650→            residue_ids,
   651→        )
   652→
   653→        results = [_pocket_result_to_out(p) for p in pockets]
   654→        task_store.update(task_id, status="completed", result=results)
   655→    except Exception as exc:
   656→        logger.exception("Pocket detection task %s failed", task_id)
   657→        task_store.update(task_id, status="failed", error=str(exc))
   658→
   659→
   660→def _run_md_simulation(
   661→    task_id: str,
   662→    target_id: str,
   663→    pdb_id: Optional[str],
   664→    n_steps: int,
   665→    dt: float,
   666→) -> None:
   667→    """Run MD simulation in a background thread."""
   668→    try:
   669→        task_store.update(task_id, status="running")
   670→
   671→        pdb_path = _resolve_pdb_path(target_id, pdb_id)
   672→        simulator = ProteinSimulator.from_pdb(str(pdb_path))
   673→        sim_result = simulator.simulate(str(pdb_path), n_steps=n_steps, dt=dt)
   674→
   675→        # Trim trajectory to avoid oversized responses
   676→        traj_frames = sim_result["trajectory"]
   677→        # Keep every 10th frame, max 50 frames
   678→        trimmed_traj = []
   679→        for i in range(0, len(traj_frames), 10):
   680→            trimmed_traj.append({
   681→                "step": traj_frames[i].get("step", i * 10),
   682→                "potential_energy": round(traj_frames[i]["potential_energy"], 4),
   683→                "kinetic_energy": round(traj_frames[i]["kinetic_energy"], 4),
   684→                "temperature": round(traj_frames[i]["temperature"], 2),
   685→            })
   686→            if len(trimmed_traj) >= 50:
   687→                break
   688→
   689→        result = {
   690→            "n_frames": len(traj_frames),
   691→            "n_frames_returned": len(trimmed_traj),
   692→            "trajectory": trimmed_traj,
   693→            "residue_ids": sim_result["residue_ids"],
   694→            "rmsf": [round(v, 4) for v in sim_result["rmsf"]],
   695→            "b_factors": [round(v, 4) for v in sim_result["b_factors"]],
   696→            "radius_of_gyration": round(sim_result["radius_of_gyration"], 4),
   697→            "n_contacts": sim_result["n_contacts"],
   698→            "final_energy": round(sim_result["final_energy"], 4),
   699→            "final_temperature": round(sim_result["final_temperature"], 2),
   700→        }
   701→        task_store.update(task_id, status="completed", result=result)
   702→    except Exception as exc:
   703→        logger.exception("MD simulation task %s failed", task_id)
   704→        task_store.update(task_id, status="failed", error=str(exc))

   707→def _run_molecule_generation(
   708→    task_id: str,
   709→    target_id: str,
   710→    pocket_center: List[float],
   711→    pocket_radius: float,
   712→    num_atoms: int,
   713→    num_molecules: int,
   714→    num_steps: int,
   715→    guidance_scale: float,
   716→) -> None:
   717→    """Run SE(3) diffusion molecule generation in a background thread."""
   718→    try:
   719→        task_store.update(task_id, status="running")
   720→
   721→        gen = MoleculeGenerator(
   722→            pocket_center=np.array(pocket_center, dtype=np.float64),
   723→            pocket_radius=pocket_radius,
   724→            num_atoms=num_atoms,
   725→        )
   726→        molecules = gen.generate(
   727→            num_molecules=num_molecules,
   728→            num_steps=num_steps,
   729→            guidance_scale=guidance_scale,
   730→        )
   731→
   732→        results = [_molecule_to_out(m) for m in molecules]
   733→        task_store.update(task_id, status="completed", result=results)
   734→    except Exception as exc:
   735→        logger.exception("Molecule generation task %s failed", task_id)
   736→        task_store.update(task_id, status="failed", error=str(exc))

    79→// --- Helper: Fitness color class ---
    80→function fitnessColor(score: number): string {
    81→  if (score > 3) return 'text-rose-500';
    82→  if (score >= 1) return 'text-amber-500';
    83→  return 'text-emerald-500';
    84→}
    85→
    86→function fitnessBg(score: number): string {
    87→  if (score > 3) return 'bg-rose-500/10 border-rose-500/20';
    88→  if (score >= 1) return 'bg-amber-500/10 border-amber-500/20';
    89→  return 'bg-emerald-500/10 border-emerald-500/20';
    90→}
    91→
    92→function fitnessBadgeVariant(score: number, isCompensatory: boolean): {
    93→  label: string;
    94→  className: string;
    95→} {
    96→  if (isCompensatory) return { label: 'Compensatory', className: 'bg-emerald-500/15 text-emerald-600 border-emerald-500/30 hover:bg-emerald-500/25' };
    97→  if (Math.abs(score) < 0.5) return { label: 'Neutral', className: 'bg-slate-500/15 text-slate-500 border-slate-500/30 hover:bg-slate-500/25' };
    98→  return { label: 'Resistance', className: 'bg-rose-500/15 text-rose-600 border-rose-500/30 hover:bg-rose-500/25' };
    99→}

   101→// --- MCTS Tree Node Renderer ---
   102→function MCTSTreeNode({ node, isLast }: { node: MCTSNode; isLast: boolean }) {
   103→  const nodeColor = node.isResistance
   104→    ? 'border-rose-400/60 bg-rose-500/5 hover:bg-rose-500/10'
   105→    : node.fitnessScore < 0
   106→      ? 'border-slate-400/40 bg-slate-500/5 hover:bg-slate-500/10'
   107→      : 'border-emerald-400/60 bg-emerald-500/5 hover:bg-emerald-500/10';
   108→
   109→  const scoreColor = node.isResistance
   110→    ? 'text-rose-500'
   111→    : node.fitnessScore < 0
   112→      ? 'text-slate-500'
   113→      : 'text-emerald-500';
   114→
   115→  const dotColor = node.isResistance
   116→    ? 'bg-rose-500'
   117→    : node.fitnessScore < 0
   118→      ? 'bg-slate-400'
   119→      : 'bg-emerald-500';
   120→
   121→  return (
   122→    <div className="flex gap-2">
   123→      {/* Vertical connector line for parent */}
   124→      <div className="relative flex flex-col items-center">
   125→        <div
   126→          className={cn(
   127→            'w-px flex-1 min-h-[20px]',
   128→            isLast ? 'bg-transparent' : 'bg-slate-300 dark:bg-slate-600'
   129→          )}
   130→        />
   131→        <div className={cn('w-2.5 h-2.5 rounded-full shrink-0 border-2 border-background', dotColor)} />
   132→        <div className="w-px flex-1 min-h-[20px] bg-slate-300 dark:bg-slate-600" />
   133→      </div>
   134→
   135→      {/* Node content + children */}
   136→      <div className="flex-1 min-w-0 pb-4">
   137→        <motion.div
   138→          initial={{ opacity: 0, x: -10 }}
   139→          animate={{ opacity: 1, x: 0 }}
   140→          transition={{ duration: 0.3, delay: node.depth * 0.1 }}
   141→          className={cn(
   142→            'rounded-lg border px-3 py-2 transition-colors',
   143→            nodeColor
   144→          )}
   145→        >
   146→          <div className="flex flex-wrap items-center gap-x-3 gap-y-1">
   147→            <span className="font-mono text-sm font-semibold text-foreground truncate">
   148→              {node.mutation}
   149→            </span>
   150→            <span className={cn('text-xs font-bold tabular-nums', scoreColor)}>
   151→              F={node.fitnessScore > 0 ? '+' : ''}{node.fitnessScore.toFixed(1)}
   152→            </span>
   153→            <span className="text-[10px] text-muted-foreground tabular-nums">
   154→              n={node.visits}
   155→            </span>
   156→          </div>
   157→        </motion.div>
   158→
   159→        {node.children.length > 0 && (
   160→          <div className="mt-1 flex flex-col">
   161→            {node.children.map((child, idx) => (
   162→              <MCTSTreeNode
   163→                key={child.id}
   164→                node={child}
   165→                isLast={idx === node.children.length - 1}
   166→              />
   167→            ))}
   168→          </div>
   169→        )}
   170→      </div>
   171→    </div>
   172→  );
   173→}

   175→// --- Epistasis interaction type styling ---
   176→function epistasisStyle(type: string) {
   177→  switch (type) {
   178→    case 'synergistic':
   179→      return { bg: 'bg-emerald-500/15', text: 'text-emerald-600', border: 'border-emerald-500/30', dot: 'bg-emerald-500' };
   180→    case 'antagonistic':
   181→      return { bg: 'bg-rose-500/15', text: 'text-rose-600', border: 'border-rose-500/30', dot: 'bg-rose-500' };
   182→    case 'additive':
   183→      return { bg: 'bg-cyan-500/15', text: 'text-cyan-600', border: 'border-cyan-500/30', dot: 'bg-cyan-500' };
   184→    default:
   185→      return { bg: 'bg-slate-500/10', text: 'text-slate-500', border: 'border-slate-500/20', dot: 'bg-slate-400' };
   186→  }
   187→}

   241→  const parseMutations = useCallback((input: string): string[] => {
   242→    return input
   243→      .replace(/,/g, '\n')
   244→      .split(/\n/)
   245→      .map((s) => s.trim().toUpperCase())

   248→
   249→  // --- Add known mutations ---
   250→  const handleAddKnown = useCallback(() => {
   251→    const known = KNOWN_MUTATIONS[selectedTargetId] ?? [];

   253→    const singles = known.filter((m) => !m.mutation.includes(':')).map((m) => m.mutation);

   255→      const existing = parseMutations(prev);
   256→      const merged = [...new Set([...existing, ...singles])];
   257→      return merged.join('\n');
   258→    });
   259→  }, [selectedTargetId, parseMutations]);

   273→    setResults(analysisResults);
   274→    setIsAnalyzing(false);
   275→    setHasRun(true);

  1148→// --- MCTS Tree helpers ---
  1149→function countNodes(node: MCTSNode): number {
  1150→  return 1 + node.children.reduce((sum, c) => sum + countNodes(c), 0);
  1151→}
  1152→
  1153→function countResistanceNodes(node: MCTSNode): number {
  1154→  return (
  1155→    (node.isResistance ? 1 : 0) +
  1156→    node.children.reduce((sum, c) => sum + countResistanceNodes(c), 0)
  1157→  );
  1158→}
  1159→
  1160→function maxDepth(node: MCTSNode): number {
  1161→  if (node.children.length === 0) return node.depth;
  1162→  return Math.max(...node.children.map(maxDepth));
  1163→}
  1164→
  1165→function bestFitness(node: MCTSNode): number {
  1166→  const childBest = node.children.length > 0
  1167→    ? Math.max(...node.children.map(bestFitness))
  1168→    : -Infinity;
  1169→  return Math.max(node.fitnessScore, childBest);
  1170→}

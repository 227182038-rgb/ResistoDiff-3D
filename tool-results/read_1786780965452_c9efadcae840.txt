     1→"""
     2→Automated bio-target dataset management system for ResistoDiff-3D.
     3→
     4→Manages a local repository of protein targets relevant to antimicrobial
     5→resistance (AMR) drug discovery.  Each target stores:
     6→
     7→  - PDB structure files (downloaded from RCSB or pre-loaded)
     8→  - Amino-acid sequences (sourced from UniProt)
     9→  - Known resistance-conferring mutations
    10→  - Pocket annotations (catalytic sites, QDR regions, etc.)
    11→
    12→Features:
    13→  - 4 pre-loaded AMR targets with REAL UniProt sequences
    14→  - RCSB PDB API integration for fetching new structures
    15→  - Auto-update mechanism that checks for new structures of watched targets
    16→  - In-memory + on-disk caching to avoid redundant downloads
    17→  - Dataset versioning with timestamps
    18→  - Export / import capability (JSON)
    19→
    20→All I/O is synchronous for simplicity; async wrappers are added in
    21→the FastAPI layer when needed.
    22→"""

    42→# ======================================================================
    43→# Data models
    44→# ======================================================================
    45→
    46→
    47→@dataclass
    48→class KnownMutation:
    49→    """A single known resistance-conferring mutation."""
    50→    position: int
    51→    wild_type: str
    52→    mutant: str
    53→    effect: str  # e.g. "ESBL", "Carbapenemase", "Fluoroquinolone resistance"
    54→    clinical_significance: str  # "high", "medium", "low"
    55→    references: List[str] = field(default_factory=list)
    56→
    57→
    58→@dataclass
    59→class PocketAnnotation:
    60→    """Catalytic / binding site annotation for a target."""
    61→    name: str  # e.g. "active_site", "zinc_binding_1", "qrdr"
    62→    residues: List[str]  # e.g. ["SER70", "LYS73", "GLU166"]
    63→    description: str = ""
    64→
    65→
    66→@dataclass
    67→class BioTarget:
    68→    """Complete bio-target record."""
    69→    target_id: str  # e.g. "tem1", "kpc2"
    70→    name: str  # e.g. "TEM-1 Beta-Lactamase"
    71→    uniprot_id: str  # e.g. "P62593"
    72→    pdb_ids: List[str]  # e.g. ["1ZG4"]
    73→    sequence: str  # one-letter AA sequence
    74→    organism: str  # e.g. "Escherichia coli"
    75→    protein_class: str  # e.g. "Beta-Lactamase (Class A)"
    76→    known_mutations: List[KnownMutation] = field(default_factory=list)
    77→    pocket_annotations: List[PocketAnnotation] = field(default_factory=list)
    78→    amr_category: str = ""  # e.g. "Beta-Lactam Resistance"
    79→    description: str = ""
    80→    last_updated: str = ""
    81→    cached_pdb_files: Dict[str, str] = field(default_factory=dict)
    82→    # pdb_id -> local path
    83→
    84→
    85→@dataclass
    86→class DatasetVersion:
    87→    """Dataset version metadata."""
    88→    version: str
    89→    timestamp: str
    90→    num_targets: int
    91→    checksum: str  # SHA-256 of the dataset JSON

   369→
   370→
   371→class DatasetManager:
   372→    """Manages the bio-target dataset on disk and in memory.
   373→
   374→    The manager:
   375→
   376→    1. Initialises the dataset directory on first use (creates folders, writes
   377→       the pre-loaded targets as JSON).
   378→    2. Provides CRUD access to targets via ``target_id``.
   379→    3. Can fetch new PDB files from the RCSB PDB REST API.
   380→    4. Supports dataset versioning, export, and import.
   381→
   382→    Parameters
   383→    ----------
   384→    dataset_path : Path
   385→        Root directory for all target data.
   386→    """
   387→
   388→    _instance: Optional[DatasetManager] = None
   389→    _rcsb_cache: Dict[str, Tuple[float, Any]] = {}
   390→    _cache_ttl: float = 3600.0
   391→
   392→    def __init__(self, dataset_path: Optional[Path] = None) -> None:
   393→        settings = get_settings()
   394→        self.base_path: Path = dataset_path or settings.dataset_path
   395→        self.targets: Dict[str, BioTarget] = {}
   396→        self.versions: List[DatasetVersion] = []
   397→        self._initialised = False
   398→
   399→    # ------------------------------------------------------------------
   400→    # Initialisation
   401→    # ------------------------------------------------------------------
   402→
   403→    def initialise(self) -> None:
   404→        """Create directory structure and load pre-loaded targets."""
   405→        if self._initialised:
   406→            return
   407→
   408→        self.base_path.mkdir(parents=True, exist_ok=True)
   409→        (self.base_path / "pdbs").mkdir(exist_ok=True)
   410→        (self.base_path / "versions").mkdir(exist_ok=True)
   411→
   412→        # Load or create the targets database
   413→        db_path = self.base_path / "targets.json"
   414→        if db_path.exists():
   415→            self._load_from_disk(db_path)
   416→            logger.info("Loaded %d targets from %s", len(self.targets), db_path)
   417→        else:
   418→            self._seed_preloaded()
   419→            self._save_to_disk(db_path)
   420→            logger.info("Seeded %d pre-loaded targets", len(self.targets))
   421→
   422→        # Load version history
   423→        version_path = self.base_path / "versions" / "history.json"
   424→        if version_path.exists():
   425→            with open(version_path, "r") as f:
   426→                data = json.load(f)
   427→                self.versions = [DatasetVersion(**v) for v in data]
   428→
   429→        self._initialised = True
   430→
   431→    def _seed_preloaded(self) -> None:
   432→        """Populate self.targets with the 4 pre-loaded AMR targets."""
   433→        now = datetime.now(timezone.utc).isoformat()
   434→        for target in PRELOADED_TARGETS:
   435→            target.last_updated = now
   436→            self.targets[target.target_id] = target
   437→
   438→    def _load_from_disk(self, db_path: Path) -> None:
   439→        """Load targets from a JSON file."""
   440→        with open(db_path, "r") as f:
   441→            data = json.load(f)

   497→    def get_pdb_path(self, target_id: str, pdb_id: str) -> Optional[Path]:
   498→        """Return the local PDB file path if it exists."""
   499→        self._ensure_init()
   500→        target = self.targets.get(target_id)
   501→        if target is None:
   502→            return None
   503→        cached = target.cached_pdb_files.get(pdb_id.upper())
   504→        if cached:
   505→            p = Path(cached)
   506→            if p.exists():
   507→                return p
   508→        # Default location
   509→        default_path = self.base_path / "pdbs" / f"{pdb_id.upper()}.pdb"
   510→        if default_path.exists():
   511→            return default_path
   512→        return None

   514→    def fetch_pdb_from_rcsb(self, pdb_id: str) -> Optional[Path]:
   515→        """Download a PDB file from RCSB and cache it locally.
   516→
   517→        Parameters
   518→        ----------
   519→        pdb_id : str
   520→            4-character PDB identifier (e.g. ``"1ZG4"``).
   521→
   522→        Returns
   523→        -------
   524→        Path or None
   525→            Local path to the downloaded file, or *None* on failure.
   526→        """
   527→        pdb_id = pdb_id.upper()
   528→        cache_key = f"pdb_{pdb_id}"
   529→
   530→        # Check in-memory cache
   531→        if cache_key in self._rcsb_cache:
   532→            ts, val = self._rcsb_cache[cache_key]
   533→            if time.time() - ts < self._cache_ttl:
   534→                if val is not None:
   535→                    return Path(val)
   536→                return None
   537→
   538→        local_path = self.base_path / "pdbs" / f"{pdb_id}.pdb"
   539→        if local_path.exists():
   540→            self._rcsb_cache[cache_key] = (time.time(), str(local_path))
   541→            return local_path
   542→
   543→        # Fetch from RCSB PDB
   544→        settings = get_settings()
   545→        url = f"{settings.rcsb_base_url}/{pdb_id}"
   546→        try:
   547→            response = httpx.get(
   548→                url,
   549→                headers={"Accept": "application/json"},
   550→                timeout=30.0,
   551→                follow_redirects=True,
   552→            )
   553→            response.raise_for_status()
   554→            data = response.json()
   555→        except Exception as exc:
   556→            logger.warning("Failed to fetch %s from RCSB: %s", pdb_id, exc)
   557→            self._rcsb_cache[cache_key] = (time.time(), None)
   558→            return None
   559→
   560→        # Extract PDB file data
   561→        pdb_file_data = None
   562→        # Try the polymer entity for atom coordinates
   563→        if "struct" in data:
   564→            title = data.get("struct", {}).get("title", "")
   565→            logger.debug("PDB %s: %s", pdb_id, title)
   566→
   567→        # Download the actual PDB file from the PDB format endpoint
   568→        pdb_file_url = f"https://files.rcsb.org/download/{pdb_id}.pdb"
   569→        try:
   570→            pdb_response = httpx.get(pdb_file_url, timeout=30.0, follow_redirects=True)
   571→            pdb_response.raise_for_status()
   572→            pdb_file_data = pdb_response.text
   573→        except Exception as exc:
   574→            logger.warning("Failed to download PDB file for %s: %s", pdb_id, exc)
   575→            self._rcsb_cache[cache_key] = (time.time(), None)
   576→            return None
   577→
   578→        local_path.write_text(pdb_file_data)
   579→        logger.info("Downloaded PDB %s to %s", pdb_id, local_path)
   580→        self._rcsb_cache[cache_key] = (time.time(), str(local_path))
   581→        return local_path

   583→    def fetch_target_pdbs(self, target_id: str) -> Dict[str, str]:
   584→        """Fetch all known PDB files for a target.
   585→
   586→        Returns a mapping {pdb_id: local_path_str}.
   587→        """
   588→        self._ensure_init()
   589→        target = self.targets.get(target_id)
   590→        if target is None:
   591→            return {}
   592→
   593→        results: Dict[str, str] = {}
   594→        for pdb_id in target.pdb_ids:
   595→            path = self.fetch_pdb_from_rcsb(pdb_id)
   596→            if path is not None:
   597→                results[pdb_id] = str(path)
   598→                target.cached_pdb_files[pdb_id] = str(path)
   599→
   600→        target.last_updated = datetime.now(timezone.utc).isoformat()
   601→        self._save_to_disk()
   602→        return results

   608→    def check_for_updates(self, target_id: str) -> Dict[str, Any]:
   609→        """Check RCSB for new structures related to a target's UniProt ID.
   610→
   611→        Returns a dict with status and any newly discovered PDB IDs.
   612→        """
   613→        self._ensure_init()
   614→        target = self.targets.get(target_id)
   615→        if target is None:
   616→            return {"status": "error", "message": f"Target '{target_id}' not found"}
   617→
   618→        cache_key = f"update_{target_id}"
   619→        if cache_key in self._rcsb_cache:
   620→            ts, val = self._rcsb_cache[cache_key]
   621→            if time.time() - ts < self._cache_ttl:
   622→                return val  # type: ignore[return-value]
   623→
   624→        # Query RCSB for structures mapping to this UniProt accession
   625→        # Query RCSB for structures mapping to this UniProt accession
   626→        payload = json.dumps({
   627→            "query": {
   628→                "type": "terminal",
   629→                "service": "text",
   630→                "parameters": {"value": target.uniprot_id},
   631→            },
   632→            "return_type": "entry",
   633→            "request_options": {
   634→                "results_content": ["experimental"],
   635→                "return_all_hits": True,
   636→            },
   637→        })
   638→        query_url = "https://search.rcsb.org/rcsbsearch/v2/query?json=" + payload
   639→        new_pdbs: List[str] = []
   640→        try:
   641→            resp = httpx.get(query_url, timeout=30.0)
   642→            resp.raise_for_status()
   643→            result = resp.json()
   644→            hit_ids = set(
   645→                hit["identifier"] for hit in result.get("result_set", [])
   646→            )
   647→            existing = set(target.pdb_ids)
   648→            new_pdbs = sorted(hit_ids - existing)
   649→        except Exception as exc:
   650→            logger.warning("Update check for %s failed: %s", target_id, exc)
   651→            result_data = {
   652→                "status": "error",
   653→                "message": str(exc),
   654→                "target_id": target_id,
   655→                "new_pdb_ids": [],
   656→            }
   657→            self._rcsb_cache[cache_key] = (time.time(), result_data)
   658→            return result_data
   659→
   660→        result_data = {
   661→            "status": "ok",
   662→            "target_id": target_id,
   663→            "target_name": target.name,
   664→            "existing_pdb_ids": sorted(target.pdb_ids),
   665→            "new_pdb_ids": new_pdbs,
   666→            "total_structures": len(target.pdb_ids) + len(new_pdbs),
   667→            "checked_at": datetime.now(timezone.utc).isoformat(),
   668→        }
   669→        self._rcsb_cache[cache_key] = (time.time(), result_data)
   670→        return result_data
   671→
   672→    def sync_all(self) -> Dict[str, Any]:
   673→        """Check all targets for updates and download new structures.
   674→
   675→        Returns a summary dict.
   676→        """
   677→        self._ensure_init()
   678→        summary = {
   679→            "status": "ok",
   680→            "targets_checked": len(self.targets),
   681→            "updates": [],
   682→            "total_new_pdbs": 0,
   683→            "sync_timestamp": datetime.now(timezone.utc).isoformat(),
   684→        }
   685→
   686→        for tid in self.targets:
   687→            update = self.check_for_updates(tid)
   688→            new_ids = update.get("new_pdb_ids", [])
   689→            if new_ids:
   690→                downloaded = self._add_new_pdbs(tid, new_ids)
   691→                summary["updates"].append({
   692→                    "target_id": tid,
   693→                    "new_pdb_ids": new_ids,
   694→                    "downloaded": downloaded,
   695→                })
   696→                summary["total_new_pdbs"] += len(downloaded)
   697→
   698→        # Bump version
   699→        self._bump_version()
   700→        return summary
   701→
   702→    def _add_new_pdbs(self, target_id: str, pdb_ids: List[str]) -> List[str]:
   703→        """Add new PDB IDs to a target and download them."""
   704→        target = self.targets.get(target_id)
   705→        if target is None:
   706→            return []
   707→
   708→        downloaded: List[str] = []
   709→        for pid in pdb_ids:
   710→            path = self.fetch_pdb_from_rcsb(pid)
   711→            if path is not None:
   712→                target.pdb_ids.append(pid)
   713→                target.cached_pdb_files[pid] = str(path)
   714→                downloaded.append(pid)
   715→
   716→        if downloaded:
   717→            target.last_updated = datetime.now(timezone.utc).isoformat()
   718→            self._save_to_disk()
   719→            # Invalidate the update cache so next check fetches fresh data
   720→            cache_key = f"update_{target_id}"
   721→            self._rcsb_cache.pop(cache_key, None)
   722→
   723→        return downloaded

   729→    def _bump_version(self) -> str:
   730→        """Create a new dataset version snapshot."""
   731→        self._ensure_init()
   732→        version_num = len(self.versions) + 1
   733→        version_str = f"v{version_num}.{int(time.time())}"
   734→
   735→        # Compute checksum of current dataset
   736→        db_path = self.base_path / "targets.json"
   737→        checksum = ""
   738→        if db_path.exists():
   739→            h = hashlib.sha256()
   740→            h.update(db_path.read_bytes())
   741→            checksum = h.hexdigest()
   742→
   743→        version = DatasetVersion(
   744→            version=version_str,
   745→            timestamp=datetime.now(timezone.utc).isoformat(),
   746→            num_targets=len(self.targets),
   747→            checksum=checksum,
   748→        )
   749→        self.versions.append(version)
   750→
   751→        # Persist version history
   752→        version_path = self.base_path / "versions" / "history.json"
   753→        with open(version_path, "w") as f:
   754→            json.dump([asdict(v) for v in self.versions], f, indent=2)
   755→
   756→        logger.info("Dataset version %s created", version_str)
   757→        return version_str
   758→
   759→    def get_version_history(self) -> List[DatasetVersion]:
   760→        """Return all dataset versions."""
   761→        self._ensure_init()
   762→        return list(self.versions)

   764→    def get_current_version(self) -> DatasetVersion:
   765→        """Return the latest version, or create one if none exists."""
   766→        self._ensure_init()
   767→        if not self.versions:
   768→            self._bump_version()
   769→        return self.versions[-1]

   797→    def import_dataset(self, import_path: Path) -> Dict[str, Any]:
   798→        """Import targets from a JSON export file.
   799→
   800→        Merges with existing data — new targets are added, existing ones
   801→        are updated (if the import version is newer).
   802→
   803→        Returns a summary dict.
   804→        """
   805→        self._ensure_init()
   806→        if not import_path.exists():
   807→            return {"status": "error", "message": f"File not found: {import_path}"}
   808→
   809→        try:
   810→            data = json.loads(import_path.read_text())
   811→        except json.JSONDecodeError as exc:
   812→            return {"status": "error", "message": f"Invalid JSON: {exc}"}
   813→
   814→        imported = 0
   815→        updated = 0
   816→        for tid, tdict in data.get("targets", {}).items():
   817→            mutations = [KnownMutation(**m) for m in tdict.get("known_mutations", [])]
   818→            pockets = [PocketAnnotation(**p) for p in tdict.get("pocket_annotations", [])]
   819→            target = BioTarget(
   820→                target_id=tdict["target_id"],
   821→                name=tdict["name"],
   822→                uniprot_id=tdict["uniprot_id"],
   823→                pdb_ids=tdict["pdb_ids"],
   824→                sequence=tdict["sequence"],
   825→                organism=tdict["organism"],
   826→                protein_class=tdict["protein_class"],
   827→                known_mutations=mutations,
   828→                pocket_annotations=pockets,
   829→                amr_category=tdict.get("amr_category", ""),
   830→                description=tdict.get("description", ""),
   831→                last_updated=tdict.get("last_updated", ""),
   832→                cached_pdb_files=tdict.get("cached_pdb_files", {}),
   833→            )
   834→            if tid in self.targets:
   835→                self.targets[tid] = target
   836→                updated += 1
   837→            else:
   838→                self.targets[tid] = target
   839→                imported += 1
   840→
   841→        self._save_to_disk()
   842→        self._bump_version()
   843→
   844→        return {
   845→            "status": "ok",
   846→            "imported": imported,
   847→            "updated": updated,
   848→            "total_targets": len(self.targets),
   849→        }
